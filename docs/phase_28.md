## 使用示例

```rust
use world_rules::rules::science::{
    ParticlePhysicsDetailedRules,
    PlasmaPhysicsDetailedRules,
    CondensedMatterDetailedRules,
};
use world_rules::rules::core::Rule;

// 粒子物理
let particle = ParticlePhysicsDetailedRules::new();
println!("标准模型: {:?}", particle.standard_model_particles());
println!("夸克规则: {:?}", particle.quark_rules());

// 等离子体物理
let plasma = PlasmaPhysicsDetailedRules::new();
println!("聚变规则: {:?}", plasma.fusion_rules());
println!("磁约束: {:?}", plasma.magnetic_confinement());

// 凝聚态物理
let condensed = CondensedMatterDetailedRules::new();
println!("能带理论: {:?}", condensed.band_theory());
println!("超导规则: {:?}", condensed.superconductivity());
```
